Preprocessing

1. Lowercase#

def preprocessing(teks):
    # 1. Lowercase
    teks = teks.lower()

    # 2. Hapus angka
    teks = re.sub(r"\d+", " ", teks)

    # 3. Hapus tanda baca
    teks = re.sub(r"[^\w\s]", " ", teks)

    # 4. Hapus spasi berlebih
    teks = re.sub(r"\s+", " ", teks).strip()

    # 5. Tokenisasi
    tokens = teks.split()

    # 6. Stopword removal
    tokens = [kata for kata in tokens if kata not in stop_words]

    # 7. Stemming
    tokens = [stemmer.stem(kata) for kata in tokens]

    return tokens

Fungsi membuat Vector Space Model

Langkah 13#

def document_vector(tokens, model):
    vectors = []

    for word in tokens:
        if word in model.wv:
            vectors.append(model.wv[word])

    if len(vectors) == 0:
        return np.zeros(model.vector_size)

    return np.mean(vectors, axis=0)

Membuat model Word2Vec Skip-Gram#

hasil_eksperimen = []

for i, parameter in enumerate(parameter_eksperimen, start=1):

    vector_size = parameter["vector_size"]
    window = parameter["window"]

    print("=" * 60)
    print(f"EKSPERIMEN {i}")
    print(f"vector_size = {vector_size}")
    print(f"window      = {window}")

    # Membuat model Word2Vec Skip-Gram
    model = Word2Vec(
        sentences=X_train_sg.tolist(),
        vector_size=vector_size,
        window=window,
        min_count=1,
        workers=1,
        sg=1,
        seed=42
    )

    # Membuat VSM training
    X_train_vsm = np.array([
        document_vector(tokens, model)
        for tokens in X_train_sg
    ])

    # Membuat VSM testing
    X_test_vsm = np.array([
        document_vector(tokens, model)
        for tokens in X_test_sg
    ])

    # Naive Bayes
    nb = GaussianNB()

    nb.fit(
        X_train_vsm,
        y_train_sg
    )

    # Prediksi
    y_pred = nb.predict(X_test_vsm)

    # Evaluasi
    accuracy = accuracy_score(
        y_test_sg,
        y_pred
    )

    precision = precision_score(
        y_test_sg,
        y_pred,
        average="weighted",
        zero_division=0
    )

    recall = recall_score(
        y_test_sg,
        y_pred,
        average="weighted",
        zero_division=0
    )

    f1 = f1_score(
        y_test_sg,
        y_pred,
        average="weighted",
        zero_division=0
    )

    print(f"Accuracy  : {accuracy:.4f}")
    print(f"Precision : {precision:.4f}")
    print(f"Recall    : {recall:.4f}")
    print(f"F1-Score  : {f1:.4f}")

    # Simpan hasil
    hasil_eksperimen.append({
        "Eksperimen": i,
        "vector_size": vector_size,
        "window": window,
        "Accuracy": accuracy,
        "Precision": precision,
        "Recall": recall,
        "F1-Score": f1
    })
